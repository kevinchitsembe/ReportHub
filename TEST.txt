# ── Query ─────────────────────────────────────────────────────
def execute_query(connection, data_inicio, data_fim):
    base_query = """
    SELECT * FROM (
        SELECT
            UP.U_NAME,
            UR.PARTY_ID,
            UP.TITLE,
            UP.MOBILENO,
            UP.FIRSTNAME,
            UP.LASTNAME,
            UP.EMAIL,
            FL.first_day,
            TO_CHAR(UP.LAST_LOGGEDIN_DATETIME, 'YY.MM.DD HH24:MI:SS') LAST_LOGGEDIN_DATE
        FROM OBDXPPRD.DIGX_UM_USERPROFILE UP
        JOIN OBDXPPRD.DIGX_UM_USERPARTY_RELATION UR
            ON UP.U_NAME = UR.USER_ID
        JOIN (
            SELECT USER_ID,
                   USER_TYPE,
                   MIN(TRUNC(CREATION_DATE)) AS first_day
            FROM OBDXPPRD.DIGX_AL_AUDIT_LOGGING
            WHERE ACTIVITY = 'Login Token Generation'
              AND STATUS = 'SUCCESS'
              AND USER_TYPE = 'corporateuser'
            GROUP BY USER_ID, USER_TYPE
        ) FL ON UP.U_NAME = FL.USER_ID
        WHERE FL.first_day BETWEEN TO_DATE(:start_date, 'YYYYMMDD')
                                AND TO_DATE(:end_date,   'YYYYMMDD')
        ORDER BY FL.first_day
    ) sub1
    ORDER BY 8 DESC
    """

    try:
        df = pd.read_sql_query(
            base_query,
            connection,
            params={"start_date": data_inicio, "end_date": data_fim}
        )
        print(f"📋 {len(df)} registo(s) encontrado(s).")
        return df

    except Exception as e:
        print(f"❌ Erro ao executar query: {e}")
        return None


# ── Guardar Excel ─────────────────────────────────────────────
def save_to_excel(df, data_fim, pasta_destino):
    script_name = os.path.splitext(os.path.basename(sys.argv[0]))[0]
    filename    = f"{script_name}_{data_fim}.xlsx"
    caminho     = os.path.join(pasta_destino, filename)

    try:
        with pd.ExcelWriter(caminho, engine='openpyxl') as writer:
            df.to_excel(writer, sheet_name='Dados', index=False)

            worksheet = writer.sheets['Dados']
            for col_idx, column in enumerate(df.columns):
                try:
                    col_width = max(
                        df[column].astype(str).map(len).max(),
                        len(column)
                    )
                except Exception:
                    col_width = len(column)

                col_letter = (
                    chr(65 + col_idx)
                    if col_idx < 26
                    else chr(64 + col_idx // 26) + chr(65 + col_idx % 26)
                )
                worksheet.column_dimensions[col_letter].width = min(col_width + 2, 50)

        print(f"✅ Ficheiro guardado em: {caminho}")

    except Exception as e:
        print(f"❌ Erro ao guardar ficheiro Excel: {e}")
        sys.exit(1)
